namespace = tno_production_events



tno_production_tech_events.2 = {
	type = country_event

	trigger = {
		has_technology_researched = dynamite
		any_scope_building = {
			OR = {
				is_building_type = building_munition_plants
				is_building_type = building_iron_mine
				is_building_type = building_coal_mine
				is_building_type = building_lead_mine
				is_building_type = building_sulfur_mine
			}
		}
		any_scope_building = {
			is_building_type = building_explosives_factory
		}
	}

	event_image = {
		video = "southamerica_factory_opening"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	duration = 3

	cooldown = { days = never_fire_again_modifier_time }

	placement = ROOT

	title = production_tech_events.102.t
	desc = production_tech_events.102.d
	flavor = production_tech_events.102.f

	immediate = {
	}

	option = { # Make national to effect chemical plants to produce more explosives.
		name = production_tech_events.102.a
		default_option = yes
		add_modifier = {
			name = dynamite_production_modifier
			days = normal_modifier_time
		}
	}

	option = { # Make national to effect mining industries
		name = production_tech_events.102.b
		add_modifier = {
			name = mining_production_modifier
			days = normal_modifier_time
		}
	}

	option = { # Make national to effect munitions plants
		name = production_tech_events.102.c
		add_modifier = {
			name = munitions_production_modifier
			days = normal_modifier_time
		}
	}
}

tno_production_tech_events.3 = { #booming consumer industry
	type = country_event

	trigger = {
		OR = {
			any_scope_building = {
				filter = {
					is_building_type = building_consumer_electronics
				}
				has_active_production_method = pm_automation_consumer_electronics_2
				cash_reserves_ratio >= 0.25
				weekly_profit > 0
				level >= 5
				occupancy >= 0.75
			}
			any_scope_building = {
				filter = {
					is_building_type = building_household_appliances
				}
				percent > 0.75
				has_active_production_method = pm_automation_household_appliances_2
				cash_reserves_ratio >= 0.25
				weekly_profit > 0
				level >= 5
				occupancy >= 0.75
			}
		}
	}

	event_image = {
		video = "southamerica_factory_opening"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	duration = 3

	cooldown = { days = never_fire_again_modifier_time }

	placement = ROOT

	title = tno_production_tech_events.3.t
	desc = tno_production_tech_events.3.d
	flavor = tno_production_tech_events.3.f

	immediate = {
	}

	option = { # consumer boost
		name = tno_production_tech_events.3.a
		default_option = yes
		add_modifier = {
			name = consumer_boom
			days = normal_modifier_time
		}
	}

	option = { #fund millitary
		name = tno_production_tech_events.3.b 
		add_modifier = {
			name = millitary_boom
			days = normal_modifier_time
		}
	}
}
