						}
					}
				}
				AND = { #Stage 4 (No restrictions whatsoever)
					game_date >= 1980.1.1
				}
	        }
		}

			base = 10
			modifier = { # The country is stronger than us
				trigger = {
					root.army_power_projection <= c:IRK.army_power_projection
				}
				add = -25
			}
			modifier = { # We are stronger than the country
				trigger = {
					root.army_power_projection >= c:IRK.army_power_projection
				}
				add = 15

		trigger = {
			var:ru_war_OSK == no
			c:OSK = {
				exists = yes
				is_adjacent_to_country = root
			}
			OR = {
				AND = { #Stage 1
					game_date >= 1966.1.1  # \/ modifier of country to invade
					has_modifier = ural

					is_diplomatic_play_target = no
					is_diplomatic_play_initiator = no
				}
				AND = { #Stage 2 (stage 1 but can attack while at war)
					game_date >= 1971.1.1  # \/ modifier of country to invade
					has_modifier = ural
				}
	            AND = { # Stage 1 Part B
	                game_date >= 1965.6.1
	                # \/ that one needs to be the modifier of the country in question, to ensure we have the same region as them
                    has_modifier = ural
	                is_diplomatic_play_target = no
					is_diplomatic_play_initiator = no
	                OR = { # We are Siberian or Far Eastern
	                    has_modifier = siberian
	                    has_modifier = far_eastern
	                }
	                c:OSK = { # The country we are attacking is also Siberian or Far Eastern
	                    OR = {
	                        has_modifier = siberian
	                        has_modifier = far_eastern
	                    }
	                }
	            }
	            AND = { #Stage 3 (attack neighboring region)
									  # \/ adjacent region
					has_modifier = western
					OR = {
						game_date >= 1975.1.1
						AND = {
							is_ai = no
							OR = {
								has_variable = ru_unite_far_east
								has_variable = ru_unite_central_sib
								has_variable = ru_unite_west_sib
								has_variable = ru_unite_west_russia
							}
						}
					}
				}
				AND = { #Stage 4 (No restrictions whatsoever)
					game_date >= 1980.1.1
				}
	        }
		}

		ai_chance = {
			base = 10
			modifier = { # The country is stronger than us
				trigger = {
					root.army_power_projection <= c:OSK.army_power_projection
				}
				add = -25
			}
			modifier = { # We are stronger than the country
				trigger = {
					root.army_power_projection >= c:OSK.army_power_projection
				}
				add = 15
			}
		}

		create_diplomatic_play = {
			name = theo
			target_country = c:OSK
			type = dp_revolution
			war = no
			escalation = 75
		}
		set_variable = {
			name = ru_opened_invasion_event
			value = no
		} 
		set_variable = { 
			name = ru_war_OSK
			value = yes 
		} 
    }
