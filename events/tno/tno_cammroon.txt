	immediate = {
	    
    }
    
    trigger = {
	}

        texture = "gfx/event_pictures/event_pictures/Africa_01.png"
    }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    option = {
		name = tno_cammroon.1.1
		default_option = yes

		add_journal_entry = { type = je_kill_yourba }
		

		trigger_event = {
			id = tno_cammroon.2
			days = 5
		}	

		

		custom_tooltip = tno_cammroon.1.1.tt.1
    }

		name = tno_cammroon.1.2
		default_option = no

		add_journal_entry = { type = je_kill_germany_africa }

		custom_tooltip = tno_cammroon.1.1.tt.2
		custom_tooltip = tno_cammroon.1.1.tt.3

		trigger_event = {
			id = tno_cammroon.3
			days = 5
		}
		 c:GER = {
            trigger_event = { id = tno_cammroon.4 days = 5 popup = yes }
        } 
		 c:KZA = {
            trigger_event = { id = tno_cammroon.4 days = 5 popup = yes }
        } 
		 c:KON = {
            trigger_event = { id = tno_cammroon.4 days = 5 popup = yes }
			
        } 
		 c:KSA = {
            trigger_event = { id = tno_cammroon.4 days = 5 popup = yes }
        } 
		

    }
}
