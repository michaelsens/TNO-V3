tno_lucknow.10 = { # Phase completed // phase check
    type = country_event
	title = tno_lucknow_debug
	desc = tno_lucknow_debug
	flavor = tno_lucknow_debug
	icon = "gfx/interface/icons/generic_icons/unused/culture_icon.dds"
	hidden = yes

	duration = 5

	immediate = {
		if = { #if tension is too high, kill them
			limit = {
				c:BHT = {
					je:je_Sphere_lucknow_conference = {
						"scripted_bar_progress(tno_lucknow_tension_progress_bar)" > 80
					}
				}
			}
			random = {
				chance = 90 #Subhas chandra bose assassination
				trigger_event = { id = tno_lucknow.12 popup = yes} #event that you died from tension
				trigger_event = { id = tno_lucknow.20 days = 240} #event about movements on the border


				c:BGL = {
					trigger_event = { id = tno_lucknow.12 popup = yes} #event that you died from tension
					trigger_event = { id = tno_lucknow.21 days = 240} #event about preparations for war
				}

				change_global_variable = { #end lucknow conference
					name = je_Sphere_lucknow_phase
					add = 50
				}

			}
		}
		if = { #if we did not just die
			limit = {
				global_var:je_Sphere_lucknow_phase < 10
			}
			random = {
				chance = je_Sphere_lucknow_conference_acceptance_actual #Acceptance chance check succeeded

				set_global_variable = { 
					name = je_Sphere_phase_successful
					value = 1
				}
				#we win

				change_global_variable = { #move it forward 100 days
					name = je_Sphere_lucknow_next_phase_date
					add = 1
				}
				trigger_event = { id = tno_lucknow.11 popup = yes} #event for phase check succeeded
				trigger_event = { id = tno_lucknow.10 days = 100} #next phase
			}

			if = { #we did not succeed with acceptance
				limit = { 
					global_var:je_Sphere_phase_successful == 0 
				}
				if = { #if we have more tries
					limit = {
						global_var:je_Sphere_lucknow_next_phase_date < 7 #if they have tried 5 times or less
					}
					change_global_variable = { #move it forward 100 days
						name = je_Sphere_lucknow_next_phase_date
						add = 1
					}
					trigger_event = { id = tno_lucknow.8 popup = yes} #event for failed phase check
					c:BGL = {
						trigger_event = { id = tno_lucknow.8 popup = yes} #event for failed phase check
					}
					trigger_event = { id = tno_lucknow.10 days = 100} #next phase
				}else = { #we've run out of tries
					trigger_event = { id = tno_lucknow.9 popup = yes} #event for them walking out
					trigger_event = { id = tno_lucknow.20 days = 240} #event about movements on the border
					c:BGL = {
						trigger_event = { id = tno_lucknow.19 popup = yes} #event for them walking out
						trigger_event = { id = tno_lucknow.21 days = 240} #event about preparations for war
					}
				}
			}
		}
		
			
	}
    
    trigger = {
    	has_journal_entry = je_Sphere_lucknow_conference
    	global_var:je_Sphere_lucknow_phase < 10
	}

	event_image = {
        texture = "gfx/event_pictures/event_pictures/lucknow_unfortunate.dds"
    }

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    option = {
		name = tno_lucknow_debug
		default_option = yes
    }
}
