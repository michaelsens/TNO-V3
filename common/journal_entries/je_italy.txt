colonize_libya = { 

    icon = "gfx/interface/icons/new_event_icons/ita_libya.dds"

    group = je_group_internal_affairs


    scripted_button = je_libya_a
    scripted_button = je_libya_b

	
    
    complete = {
        
			
				
		any_scope_state = {
            filter = {
                state_region = s:STATE_LIBYA
            }
          
             culture_percent_state = {
                target = cu:north_italian
                 value >= 0.25
                } 
        }
        any_scope_state = {
            filter = {
                state_region = s:STATE_TRIPOLI
            }
          
             culture_percent_state = {
                target = cu:north_italian
                 value >= 0.25
                } 
        }

    }

    on_complete = {

        s:STATE_TRIPOLI = {
				add_homeland = cu:north_italian
		}
        s:STATE_LIBYA = {
				add_homeland = cu:north_italian
		}
   

    }

    
    weight = 5
    
    should_be_pinned_by_default = yes
    
}
