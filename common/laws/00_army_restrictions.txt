law_encouraged_brutality = {
	group = lawgroup_army_restrictions
	icon = "gfx/interface/icons/law_icons/peasant_levies.dds"
	
	progressiveness = -50
	
	modifier = {
		country_damage_relations_speed_mult = 0.5
		country_improve_relations_speed_mult = -0.5
		country_prestige_mult = -0.1
		unit_kill_rate_add = 0.1
		unit_morale_damage_mult = 0.1
		country_infamy_decay_mult = -0.25
		country_infamy_generation_mult = 0.25
	}
	
	on_enact = {
	}
	
	possible_political_movements = {
		
	}

	pop_support = {
		value = 0
		# Aristocrats like feudalism
		add = {
			desc = "POP_ARISTOCRATS"			
			if = {
				limit = { 
					is_pop_type = aristocrats
				}
				value = 0.33
			}
		}
	}

	# AI will never return to peasant levies
	ai_will_do = {
		always = no
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { 
				has_journal_entry = je_government_petition
				has_variable = desired_law_var
                scope:law = var:desired_law_var
			}
			add = 750
		}
	}
}
