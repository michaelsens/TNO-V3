law_propaganda_press = {
	group = lawgroup_colonization
	
	icon = "gfx/interface/icons/law_icons/no_colonial_affairs.dds"
	
	progressiveness = 0	
	
	pop_support = {
		value = 0
	}

	# AI will never fully dismantle institutions
	ai_will_do = {
		always = no
	}	

	modifier = {
		state_radicals_from_sol_change_mult = -0.3
		country_authority_mult = 0.2
		state_radicals_from_discrimination_mult = 0.3
		country_legitimacy_base_add = 10
		interest_group_in_government_attraction_mult = 0.5
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		
		if = {
			limit = { 
				has_journal_entry = je_government_petition
				has_variable = desired_law_var
                scope:law = var:desired_law_var
			}
			add = 750
		}
	}
}

law_state_press_only = {
	group = lawgroup_colonization

	can_enact = {
		any_scope_state = {
			is_homeland_of_country_cultures = ROOT
			any_neighbouring_state = {
				owner = {
					is_country_type = decentralized
				}
			}
		}
	}
	
	icon = "gfx/interface/icons/law_icons/frontier_colonization.dds"
	
	progressiveness = 25

	unlocking_technologies = {
		urbanization
	}

	
	modifier = {
		state_radicals_from_sol_change_mult = -0.2
		country_authority_mult = 0.1
		state_radicals_from_discrimination_mult = 0.1
		country_legitimacy_base_add = 5
		interest_group_in_government_attraction_mult = 0.20
	}
	
	

	pop_support = {
		value = 0
		# Farmers like frontiers
		add = {
			desc = "POP_FARMERS"			
			if = {
				limit = {
					is_pop_type = farmers
				}
				value = 0.25
			}
		}
	}
	
	ai_will_do = {
		NOR = {
			has_law = law_type:law_colonial_exploitation
			has_law = law_type:law_state_press_only
		}
	}
}
